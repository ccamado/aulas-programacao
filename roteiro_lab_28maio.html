Parte A — Crie e explore um dicionário:

# EXERCÍCIO 3A — Dicionário simples aluno = { "nome": "Maria", "idade": 17, "turma": "3ºA", "nota": 8.5 } # Acessar valores pelo nome da chave print("Nome:", aluno["nome"]) print("Nota:", aluno["nota"]) # Percorrer todas as chaves e valores print("--- Ficha do aluno ---") for chave, valor in aluno.items(): print(f"{chave}: {valor}")

# EXERCÍCIO 3B — Mini-cadastro (3 alunos) alunos = [ {"nome": "João", "nota": 7.0}, {"nome": "Maria", "nota": 8.5}, {"nome": "Pedro", "nota": 5.5}, ] print("=== Relatório da Turma ===") for a in alunos: situacao = "Aprovado ✓" if a["nota"] >= 6 else "Reprovado ✗" print(f"{a['nome']} — {a['nota']} — {situacao}")
