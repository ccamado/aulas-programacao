Cenário 1

Venda de roupa

Calcula o total da compra e sorteia um cupom quando o valor permite desconto.

import random

def calcular_total(preco, quantidade):
    return preco * quantidade

cliente = input("Nome do cliente: ")
peca = input("Peça de roupa: ")
preco = float(input("Preço da peça: "))
quantidade = int(input("Quantidade: "))

total = calcular_total(preco, quantidade)
cupom = random.randint(5, 20)

print("Cliente:", cliente)
print("Peça:", peca)
print("Quantidade:", quantidade)
print("Total sem desconto:", total)
print("Cupom sorteado:", cupom, "%")

if total >= 100:
    desconto = total * cupom / 100
    total_final = total - desconto
    print("Desconto aplicado:", desconto)
    print("Total final:", total_final)
else:
    print("Compra abaixo de R$100. Sem desconto especial.")
Cenário 2

Veterinária

Calcula uma idade aproximada do pet e sorteia a senha de atendimento.

import random

def calcular_idade_humana(idade_pet):
    return idade_pet * 7

pet = input("Nome do pet: ")
tipo = input("Tipo do animal: ")
idade = int(input("Idade do pet: "))

idade_humana = calcular_idade_humana(idade)
senha = random.randint(1, 50)

print("Pet:", pet)
print("Tipo:", tipo)
print("Idade aproximada em anos humanos:", idade_humana)
print("Senha de atendimento:", senha)

if idade >= 8:
    print("Atenção: pet idoso, recomendar check-up.")
else:
    print("Pet jovem/adulto, atendimento normal.")
Cenário 3

Boletim escolar

Usa lista de notas, calcula a média e sorteia um pequeno bônus.

import random

def media(notas):
    return sum(notas) / len(notas)

aluno = input("Nome do aluno: ")
notas = []

for i in range(1, 4):
    nota = float(input(f"Nota {i}: "))
    notas.append(nota)

m = media(notas)
bonus = random.randint(0, 1)
media_final = m + bonus

print("Aluno:", aluno)
print("Notas:", notas)
print("Média:", m)
print("Bônus sorteado:", bonus)
print("Média final:", media_final)

if media_final >= 6:
    print("Situação: Aprovado")
else:
    print("Situação: Recuperação")
Cenário 4

Pizzaria

Calcula o total do pedido e sorteia um brinde para compras maiores.

import random

def calcular_pedido(valor_pizza, quantidade):
    return valor_pizza * quantidade

cliente = input("Nome do cliente: ")
sabor = input("Sabor da pizza: ")
valor = float(input("Valor da pizza: "))
quantidade = int(input("Quantidade: "))

total = calcular_pedido(valor, quantidade)
brinde = random.choice(["Refrigerante", "Borda recheada", "Cupom de desconto"])

print("Cliente:", cliente)
print("Sabor:", sabor)
print("Quantidade:", quantidade)
print("Total:", total)

if total >= 80:
    print("Pedido ganhou brinde:", brinde)
else:
    print("Pedido simples, sem brinde especial.")
Cenário 5

Oficina mecânica

Registra carro, serviço, peças e mecânico; depois calcula orçamento e prazo.

import random

def calcular_total(servico, peca):
    return servico + peca

cliente = input("Nome do cliente: ")
carro = input("Modelo do carro: ")
servico_nome = input("Serviço realizado: ")
peca_nome = input("Peça trocada: ")
mecanico = input("Nome do mecânico: ")

valor_servico = float(input("Valor do serviço: "))
valor_peca = float(input("Valor da peça: "))

total = calcular_total(valor_servico, valor_peca)
prazo = random.randint(1, 5)

ordem_servico = {
    "cliente": cliente,
    "carro": carro,
    "serviço": servico_nome,
    "peça": peca_nome,
    "mecânico": mecanico,
    "total": total
}

print("Ordem de serviço:", ordem_servico)
print("Prazo estimado:", prazo, "dias")

if total >= 500:
    print("Serviço grande. Revisar orçamento com o cliente.")
else:
    print("Serviço simples. Pode seguir com aprovação rápida.")
