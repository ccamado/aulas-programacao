notas = [] for i in range(5): nota = float(input(f"Nota {i + 1}: ")) notas.append(nota) media = sum(notas) / len(notas) maior = max(notas) aprovadas = 0 for nota in notas: if nota >= 6: aprovadas = aprovadas + 1 print("Notas:", notas) print("Média:", media) print("Maior nota:", maior) print("Notas aprovadas:", aprovadas)
